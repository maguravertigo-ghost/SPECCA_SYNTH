import { getSeq, getSongDuration, resumeAudio } from '../core/synth.js';

const fmt = s => {
    if (!isFinite(s) || s < 0) s = 0;
    const m = Math.floor(s / 60);
    return `${m}:${Math.floor(s - m * 60).toString().padStart(2, '0')}`;
};

export class Transport {
    state = 'stopped';
    rafId = null;
    seekActive = false;

	constructor({ playBtn, pauseBtn, stopBtn, timeEl, totalEl, progressEl, progressFill, onPlayStart, onStop }) {
		Object.assign(this, { playBtn, pauseBtn, stopBtn, timeEl, totalEl, progressEl, progressFill, onPlayStart, onStop });
		this._bind();
		this._render();
	}

    _bind() {
        this.playBtn?.addEventListener('click', e => { e.preventDefault(); this._togglePlay(); });
        this.pauseBtn?.addEventListener('click', e => { e.preventDefault(); this._togglePause(); });
        this.stopBtn?.addEventListener('click', e => { e.preventDefault(); this._stop(); });

        if (this.progressEl) {
            const seekFromClientX = clientX => {
                const seq = getSeq();
                if (!seq || !seq.duration) return;
                const r = this.progressEl.getBoundingClientRect();
                const ratio = Math.max(0, Math.min(1, (clientX - r.left) / r.width));
                seq.currentTime = ratio * seq.duration;
                this._render();
            };

            this.progressEl.addEventListener('pointerdown', e => {
                this.seekActive = true;
                this.progressEl.setPointerCapture?.(e.pointerId);
                seekFromClientX(e.clientX);
            });
            this.progressEl.addEventListener('pointermove', e => {
                if (this.seekActive) seekFromClientX(e.clientX);
            });
            const end = e => {
                if (!this.seekActive) return;
                this.seekActive = false;
                try { this.progressEl.releasePointerCapture?.(e.pointerId); } catch {}
                if (this.state === 'playing') this._tick();
            };
            this.progressEl.addEventListener('pointerup', end);
            this.progressEl.addEventListener('pointercancel', end);
        }
    }

	async _togglePlay() {
		const seq = getSeq();
		if (!seq) return;
		if (this.state === 'playing') return;

		await resumeAudio();
		this.onPlayStart?.();   // ← новая строка

		if (this.state === 'stopped' || seq.isFinished) {
			try { seq.currentTime = 0; } catch {}
		}
		seq.play();
		this._sync('playing');
		this._tick();
	}

    async _togglePause() {
        const seq = getSeq();
        if (!seq || this.state === 'stopped') return;

        if (seq.paused) {
            await resumeAudio();
            seq.play();
            this._sync('playing');
            this._tick();
        } else {
            seq.pause();
            this._sync('paused');
            this._render();
        }
    }

    _stop() {
        const seq = getSeq();
        if (!seq) return;
		this.onStop?.();                   // ← новая строка
        seq.pause();
        try { seq.currentTime = 0; } catch {}
        this._sync('stopped');
        this._render();
    }

    reset() {
        this._sync('stopped');
        this._render();
    }

	_sync(state) {
		this.state = state;
		const set = (btn, active) => {
			if (!btn) return;
			btn.disabled = false;            // ← добавить
			btn.setAttribute('aria-disabled', active ? 'false' : 'true');
			btn.classList.toggle('is-inactive', !active);
		};
		set(this.playBtn,  state !== 'playing');
		set(this.pauseBtn, state !== 'stopped');
		set(this.stopBtn,  state !== 'stopped');
		if (this.pauseBtn) this.pauseBtn.textContent = state === 'paused' ? 'Resume' : 'Pause';
	}

_tick() {
    if (this.rafId) { cancelAnimationFrame(this.rafId); this.rafId = null; }
    const loop = () => {
        const seq = getSeq();
        if (!seq) { this.rafId = null; return; }
        this._render();

        // Подхватываем внешний play (например, из piano roll):
        // если seq не на паузе и не завершён, а мы ещё в 'stopped'/'paused' —
        // переключаемся в 'playing'.
        if (!this.seekActive && !seq.paused && !seq.isFinished && this.state !== 'playing') {
            this._sync('playing');
        }
        // Подхватываем внешний pause:
        if (!this.seekActive && seq.paused && this.state === 'playing') {
            this._sync('paused');
        }

        if (seq.isFinished) {
            this._sync('stopped');
            this.rafId = null;
            return;
        }

        if (!this.seekActive && this.state !== 'stopped') {
            this.rafId = requestAnimationFrame(loop);
        } else {
            this.rafId = null;
        }
    };
    this.rafId = requestAnimationFrame(loop);
}

// Публичный метод для внешнего «включения» — вызывается из piano roll
syncExternal() {
    const seq = getSeq();
    if (!seq) return;
    if (!seq.paused && !seq.isFinished) {
        this._sync('playing');
        this._tick();
    }
}

    _render() {
        const seq = getSeq();
        if (!seq) return;
        const dur = getSongDuration();
        const cur = Math.min(seq.currentTime || 0, dur);
        if (this.timeEl)  this.timeEl.textContent  = fmt(cur);
        if (this.totalEl) this.totalEl.textContent = fmt(dur);
        if (this.progressFill) {
            this.progressFill.style.width = (dur > 0 ? (cur / dur) * 100 : 0) + '%';
        }
    }
}

    const $ = (id) => document.getElementById(id);
    const {
        // DOM (можно переопределить, иначе берётся из index.html)
        playBtn, pauseBtn, stopBtn,
        timeEl, totalEl, progressEl, progressFill,
        // Колбэки
        onPlayStart, onStop,
        // Устаревшие/на будущее — сейчас игнорируются, но принимаем,
        // чтобы не менять сигнатуру вызовов в main.js
        getPlayback: _getPlayback,
        onSeek: _onSeek,
    } = opts;

    return new Transport({
        playBtn:       playBtn       ?? $('playBtn'),

        stopBtn:       stopBtn       ?? $('stopBtn'),
        timeEl:        timeEl        ?? $('timeNow'),

    });
}
